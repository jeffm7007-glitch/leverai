 <script>
 (function() {
 const projects = {
 drain: { baseLow: 150, baseHigh: 300 },
 heater: { baseLow: 450, baseHigh: 900 },
 leak: { baseLow: 200, baseHigh: 400 },
 installation: { baseLow: 350, baseHigh: 700 }
 };

 const sizeMultipliers = {
 small: 1.0,
 medium: 1.8,
 large: 3.2
 };

 const slider = document.getElementById('qp-material-slider');
 const budgetVal = document.getElementById('qp-budget-val');
 const calcBtn = document.getElementById('qp-calc-trigger');
 const priceDisplay = document.getElementById('qp-price-display');

 slider.addEventListener('input', function() {
 budgetVal.textContent = '$' + this.value;
 });

 function updateCalculation() {
 const selectedProject = document.querySelector('input[name="qp_project"]:checked').value;
 const selectedSize = document.querySelector('input[name="qp_size"]:checked').value;
 const materialBudget = parseFloat(slider.value) || 0;

 const pData = projects[selectedProject];
 const mult = sizeMultipliers[selectedSize];

 let minTotal = Math.round((pData.baseLow * mult) + materialBudget);
 let maxTotal = Math.round((pData.baseHigh * mult) + (materialBudget * 1.25));

 // Format numbers with commas
 const formattedMin = minTotal.toLocaleString();
 const formattedMax = maxTotal.toLocaleString();

 priceDisplay.textContent = `$${formattedMin} - $${formattedMax}*`;
 }

 calcBtn.addEventListener('click', updateCalculation);

 // Initial calculation setup
 updateCalculation();
 })();
 </script>
