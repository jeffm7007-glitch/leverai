var plmSel={name:'Emergency Repair',base:350,hours:2};
function plmPick(el){
  document.querySelectorAll('.plm-jc').forEach(function(c){
    c.style.border='2px solid #e2e8f0';c.style.background='#fff';c.style.boxShadow='none';
  });
  el.style.border='2px solid #1e3a5f';el.style.background='#f0f7ff';el.style.boxShadow='0 0 0 3px rgba(30,58,95,0.1)';
  plmSel={name:el.dataset.name,base:parseFloat(el.dataset.base),hours:parseFloat(el.dataset.hours)};
  document.getElementById('plm-hrs-in').value=plmSel.hours;
  document.getElementById('plm-hrs-sl').value=plmSel.hours;
}
function plmCalc(){
  var rate=parseFloat(document.getElementById('plm-rate-in').value)||0;
  var hours=parseFloat(document.getElementById('plm-hrs-in').value)||0;
  var mats=parseFloat(document.getElementById('plm-mat-in').value)||0;
  var trip=parseFloat(document.getElementById('plm-trip-in').value)||0;
  var taxRate=parseFloat(document.getElementById('plm-tax-in').value)||0;
  var labor=rate*hours;
  var sub=labor+mats+trip;
  var tax=sub*(taxRate/100);
  var total=sub+tax;
  document.getElementById('plm-total').textContent='$'+Math.round(total).toLocaleString();
  document.getElementById('plm-job').textContent=plmSel.name;
  document.getElementById('plm-labor').textContent='$'+Math.round(labor)+' ('+hours+' hrs @ $'+rate+'/hr)';
  document.getElementById('plm-mat-d').textContent='$'+mats;
  document.getElementById('plm-trip-d').textContent='$'+trip;
  document.getElementById('plm-sub').textContent='$'+Math.round(sub).toLocaleString();
  document.getElementById('plm-tax-d').textContent='$'+Math.round(tax)+' ('+taxRate+'%)';
  document.getElementById('plm-res').style.display='block';
}
